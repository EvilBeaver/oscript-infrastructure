# robots.txt хаба пакетов OneScript — hub.oscript.io.
# Отдаёт nginx из web/nginx/robots/hub.oscript.io.txt, в хаб запрос не уходит.
#
# Роботам открыты витрина и страницы пулов и пакетов: /, /packages, /pools/{пул},
# /pools/{пул}/packages/{пакет}. Закрыто машинное и служебное. Под /pools/ живут и страницы,
# и выдача, поэтому префикс целиком не закрыт, а машинные адреса названы по одному.
# «$» — конец адреса, «?» — тот же адрес с параметрами, «/» на конце — всё, что под ним.
# «*» проходит и через «/»: правила /pools/*/push и /pools/*/changes закрывают заодно
# страницы пакетов с именами push и changes, а /*/settings, /*/versions, /*/deps — пулы
# и пакеты с такими именами.

User-agent: *
Crawl-delay: 5

# выдача пакетов и API — для opm и зеркал, не для поиска

Disallow: /download/
Disallow: /dev-channel/
Disallow: /pools/*/download/
Disallow: /pools/*/channels/
Disallow: /pools/*/changes$
Disallow: /pools/*/changes?
Disallow: /changes$
Disallow: /changes?

# публикация

Disallow: /push?
Disallow: /pools/*/push$
Disallow: /pools/*/push?

Disallow: /confirm-email?
